import 'package:flutter/material.dart';

class ChatRoomScreen extends StatefulWidget {
  final String userId;
  final String userName;

  const ChatRoomScreen({Key? key, required this.userId, required this.userName}) : super(key: key);

  @override
  _ChatRoomScreenState createState() => _ChatRoomScreenState();
}

class _ChatRoomScreenState extends State<ChatRoomScreen> {
  final FirebaseFirestore _db = FirebaseFirestore.instance;
  final TextEditingController _messageController = TextEditingController();

  void _sendMessage() async {
    String text = _messageController.text.trim();
    if (text.isEmpty) return;

    try {
      await _db.collection('chat_messages').add({
        'userId': widget.userId,
        'userName': widget.userName,
        'message': text,
        'timestamp': FieldValue.serverTimestamp(),
      });
      _messageController.clear();
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('خطأ في إرسال الرسالة: ${e.toString()}')),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('غرفة الدردشة الحية - منصة مودة'),
        backgroundColor: Colors.black,
      ),
      body: Column(
        children: [
          // عرض رسائل الغرفة الحية بشكل لحظي
          Expanded(
            child: StreamBuilder<QuerySnapshot>(
              stream: _db.collection('chat_messages').orderBy('timestamp', descending: true).snapshots(),
              builder: (context, snapshot) {
                if (!snapshot.hasData) {
                  return const Center(child: CircularProgressIndicator());
                }

                var docs = snapshot.data!.docs;
                if (docs.isEmpty) {
                  return const Center(child: Text('لا توجد رسائل بعد، كن أول المطارحين!'));
                }

                return ListView.builder(
                  reverse: true,
                  itemCount: docs.length,
                  itemBuilder: (context, index) {
                    var data = docs[index].data() as Map<String, dynamic>;
                    String senderName = data['userName'] ?? 'عضو مجهول';
                    String messageText = data['message'] ?? '';
                    bool isMe = data['userId'] == widget.userId;

                    return Container(
                      margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
                      alignment: isMe ? Alignment.centerRight : Alignment.centerLeft,
                      child: Container(
                        padding: const EdgeInsets.all(12),
                        decoration: BoxDecoration(
                          color: isMe ? Colors.pink[100] : Colors.grey[200],
                          borderRadius: BorderRadius.circular(12),
                        ),
                        child: Column(
                          crossAxisAlignment: isMe ? CrossAxisAlignment.end : CrossAxisAlignment.start,
                          children: [
                            Text(
                              senderName,
                              style: TextStyle(fontSize: 12, fontWeight: FontWeight.bold, color: Colors.pink[800]),
                            ),
                            const SizedBox(height: 4),
                            Text(
                              messageText,
                              style: const TextStyle(fontSize: 16),
                            ),
                          ],
                        ),
                      ),
                    );
                  },
                );
              },
            ),
          ),
          // شريط كتابة وإرسال الرسائل
          Container(
            padding: const EdgeInsets.all(8),
            color: Colors.white,
            child: Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: _messageController,
                    decoration: const InputDecoration(
                      hintText: 'اكتب رسالتك هنا...',
                      border: OutlineInputBorder(),
                      contentPadding: EdgeInsets.symmetric(horizontal: 12, vertical: 8),
                    ),
                  ),
                ),
                const SizedBox(width: 8),
                IconButton(
                  icon: const Icon(Icons.send, color: Colors.pink),
                  onPressed: _sendMessage,
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}

import 'package:flutter/material.dart';
import '../services/auth_service.dart';
import 'main_user_screen.dart';

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final TextEditingController _emailController = TextEditingController();
  final TextEditingController _passwordController = TextEditingController();
  final AuthService _authService = AuthService();
  bool _isLoading = false;
  bool _isSignUpMode = false;

  void _submitForm() async {
    final email = _emailController.text.trim();
    final password = _passwordController.text.trim();

    if (email.isEmpty || password.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('يرجى إدخال البريد الإلكتروني وكلمة المرور')),
      );
      return;
    }

    setState(() => _isLoading = true);

    try {
      if (_isSignUpMode) {
        // إنشاء حساب جديد للأعضاء الجدد
        await _authService.signUpWithEmail(email, password);
      } else {
        // تسجيل الدخول لحساب قائم
        await _authService.signInWithEmail(email, password);
      }

      if (mounted) {
        // التوجيه الصحيح إلى الشاشة الرئيسية للمستخدم (وليس المتجر)
        Navigator.pushReplacement(
          context,
          MaterialPageRoute(builder: (context) => const MainUserScreen()),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('حدث خطأ: ${e.toString()}')),
        );
      }
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(_isSignUpMode ? 'إنشاء حساب جديد' : 'تسجيل الدخول - منصة مودة'),
        centerTitle: true,
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: SingleChildScrollView(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const SizedBox(height: 40),
              TextField(
                controller: _emailController,
                decoration: const InputDecoration(
                  labelText: 'البريد الإلكتروني',
                  border: OutlineInputBorder(),
                ),
                keyboardType: TextInputType.emailAddress,
              ),
              const SizedBox(height: 16),
              TextField(
                controller: _passwordController,
                decoration: const InputDecoration(
                  labelText: 'كلمة المرور',
                  border: OutlineInputBorder(),
                ),
                obscureText: true,
              ),
              const SizedBox(height: 24),
              _isLoading
                  ? const CircularProgressIndicator()
                  : ElevatedButton(
                      onPressed: _submitForm,
                      style: ElevatedButton.styleFrom(
                        minimumSize: const Size.fromHeight(50),
                      ),
                      child: Text(_isSignUpMode ? 'إنشاء الحساب الان' : 'دخول'),
                    ),
              const SizedBox(height: 16),
              TextButton(
                onPressed: () {
                  setState(() {
                    _isSignUpMode = !_isSignUpMode;
                  });
                },
                child: Text(_isSignUpMode
                    ? 'لديك حساب بالفعل؟ سجل الدخول هنا'
                    : 'ليس لديك حساب؟ اضغط هنا لإنشاء حساب جديد'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class UserProfileScreen extends StatefulWidget {

  const UserProfileScreen({Key? key, required this.userId}) : super(key: key);

  @override
  _UserProfileScreenState createState() => _UserProfileScreenState();
}

class _UserProfileScreenState extends State<UserProfileScreen> {
  final FirebaseFirestore _db = FirebaseFirestore.instance;

  // متحكمات الحقول الشخصية التي تحدث عنها
  final TextEditingController _nameController = TextEditingController();
  final TextEditingController _ageController = TextEditingController();
  final TextEditingController _professionController = TextEditingController();
  
  String _maritalStatus = 'أعزب / عزباء';
  String _relationshipGoal = 'زواج جاد وعلاقة مستقرة';
  bool _isLoading = true;

  @override
  void initState() {
    super.initState();
    _loadUserData();
  }

  // جلب بيانات المستخدم الحالية من قاعدة البيانات
  Future<void> _loadUserData() async {
    DocumentSnapshot doc = await _db.collection('users').doc(widget.userId).get();
    if (doc.exists) {
      var data = doc.data() as Map<String, dynamic>;
      setState(() {
        _nameController.text = data['fullName'] ?? '';
        _ageController.text = data['age']?.toString() ?? '';
        _professionController.text = data['profession'] ?? '';
        _maritalStatus = data['maritalStatus'] ?? 'أعزب / عزباء';
        _relationshipGoal = data['relationshipGoal'] ?? 'زواج جاد وعلاقة مستقرة';
        _isLoading = false;
      });
    } else {
      setState(() {
        _isLoading = false;
      });
    }
  }

  // حفظ وتحديث المعلومات الشخصية في Firebase
  Future<void> _saveProfileChanges() async {
    setState(() => _isLoading = true);
    await _db.collection('users').doc(widget.userId).update({
      'fullName': _nameController.text.trim(),
      'age': int.tryParse(_ageController.text.trim()) ?? 25,
      'profession': _professionController.text.trim(),
      'maritalStatus': _maritalStatus,
      'relationshipGoal': _relationshipGoal,
      'updatedAt': FieldValue.serverTimestamp(),
    });
    setState(() => _isLoading = false);

    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('تم تحديث معلومات الملف الشخصي بنجاح')),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('الملف الشخصي وتعديل المعلومات'),
        backgroundColor: Colors.pink,
      ),
      body: _isLoading
          ? const Center(child: CircularProgressIndicator())
          : Padding(
              padding: const EdgeInsets.all(16.0),
              child: ListView(
                children: [
                  const Center(
                    child: CircleAvatar(
                      radius: 50,
                      backgroundColor: Colors.pinkAccent,
                      child: Icon(Icons.person, size: 50, color: Colors.white),
                    ),
                  ),
                  const SizedBox(height: 20),
                  TextField(
                    controller: _nameController,
                    decoration: const InputDecoration(
                      labelText: 'الاسم الكامل',
                      border: OutlineInputBorder(),
                      prefixIcon: Icon(Icons.person_outline),
                    ),
                  ),
                  const SizedBox(height: 15),
                  TextField(
                    controller: _ageController,
                    keyboardType: TextInputType.number,
                    decoration: const InputDecoration(
                      labelText: 'العمر',
                      border: OutlineInputBorder(),
                      prefixIcon: Icon(Icons.cake_outlined),
                    ),
                  ),
                  const SizedBox(height: 15),
                  TextField(
                    controller: _professionController,
                    decoration: const InputDecoration(
                      labelText: 'المهنة',
                      border: OutlineInputBorder(),
                      prefixIcon: Icon(Icons.work_outline),
                    ),
                  ),
                  const SizedBox(height: 15),
                  DropdownButtonFormField<String>(
                    value: _maritalStatus,
                    decoration: const InputDecoration(
                      labelText: 'الحالة الاجتماعية',

                      DropdownMenuItem(value: 'أرمل / أرملة', child: Text('أرمل / أرملة')),
                    ],

  // دالة مسؤولة عن التحقق من الرصيد وخصم الكوينز عند الشراء
  Future<bool> _purchaseItem(int cost, String description) async {
    try {
      DocumentReference userRef = _db.collection('users').doc(widget.userId);
      var userDoc = await userRef.get();
      if (!userDoc.exists) return false;

      var userData = userDoc.data() as Map<String, dynamic>;
      int currentCoins = userData['coins'] ?? 0;

      if (currentCoins < cost) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('رصيدك غير كافٍ لإتمام عملية الشراء!')),
        );
        return false;
      }

      // خصم الكوينز وتوثيق العملية في سجل المعاملات
      await _db.runTransaction((transaction) async {
        transaction.update(userRef, {'coins': currentCoins - cost});
        transaction.set(_db.collection('transactions').doc(), {
          'userId': widget.userId,
          'type': 'purchase',
          'description': description,
          'cost': cost,
          'timestamp': FieldValue.serverTimestamp(),
        });
      });

      return true;
    } catch (e) {
      return false;
    }
  }

import 'package:firebase_auth/firebase_auth.dart';

class AuthService {
  final FirebaseAuth _auth = FirebaseAuth.instance;

  // الحصول على المستخدم الحالي
  User? get currentUser => _auth.currentUser;

  // تسجيل الدخول بالبريد وكلمة المرور
  Future<User?> signInWithEmail(String email, String password) async {
    try {
      UserCredential result = await _auth.signInWithEmailAndPassword(
        email: email,
        password: password,
      );
      return result.user;
    } catch (e) {
      print('خطأ في تسجيل الدخول: $e');
      return null;
    }
  }

  // إنشاء حساب جديد بالبريد وكلمة المرور
  Future<User?> signUpWithEmail(String email, String password) async {
    try {
      UserCredential result = await _auth.createUserWithEmailAndPassword(
        email: email,
        password: password,
      );
      return result.user;
    } catch (e) {
      print('خطأ في إنشاء الحساب: $e');
      return null;
    }
  }

  // تسجيل الخروج
  Future<void> signOut() async {
    await _auth.signOut();
  }
}

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';

class CoinSystemService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;
  final FirebaseAuth _auth = FirebaseAuth.instance;

  Future<int> getUserCoins() async {
    try {
      User? user = _auth.currentUser;
      if (user == null) return 0;

      DocumentSnapshot doc = await _firestore.collection('users').doc(user.uid).get();
      if (doc.exists && doc.data() != null) {
        Map<String, dynamic> data = doc.data() as Map<String, dynamic>;
        return data['coins'] ?? data['coinsBalance'] ?? 0;
      }
    } catch (e) {
      print('خطأ في جلب الأرصدة: $e');
    }
    return 0;
  }

  Future<void> updateCoins(int newBalance) async {
    try {
      User? user = _auth.currentUser;
      if (user == null) return;

      await _firestore.collection('users').doc(user.uid).update({
        'coins': newBalance,
      });
    } catch (e) {
      print('خطأ في تحديث الأرصدة: $e');
    }
  }
}
import 'package:cloud_firestore/cloud_firestore.dart';
import '../models/user_model.dart';

class DatabaseService {
  final FirebaseFirestore _db = FirebaseFirestore.instance;

  // حفظ بيانات المستخدم الجديد مع منح مكافأة المرحلة الأولى (500 كوينز لتوقيع البريد)
  Future<void> saveNewUser(UserModel user) async {
    final userWithBonus = UserModel(
      uid: user.uid,
      email: user.email,
      fullName: user.fullName,
      gender: user.gender,
      bio: user.bio,
      profileImageUrl: user.profileImageUrl,
      isEmailVerified: true,
      coinsBalance: 500, // مكافأة المرحلة الأولى
      activeTitle: 'عضو مؤسس',
      role: 'user',
    );

    await _db.collection('users').doc(user.uid).set(userWithBonus.toMap());
  }

  // تحديث حالة التوثيق البيومتري ومنح المرحلة الثانية (1500 كوينز إضافية ليصبح المجموع 2000)
  Future<void> verifyUserBiometric(String uid, int currentCoins) async {
    await _db.collection('users').doc(uid).update({
      'isBiometricVerified': true,
      'coinsBalance': currentCoins + 1500, // مكافأة المرحلة الثانية
    });
  }

  // جلب بيانات المستخدم
  Future<UserModel?> getUserData(String uid) async {
    DocumentSnapshot doc = await _db.collection('users').doc(uid).get();
    if (doc.exists) {
      return UserModel.fromMap(doc.data() as Map<String, dynamic>, doc.id);
    }
    return null;
  }
}
